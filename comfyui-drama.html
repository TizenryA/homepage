Chapter 03

🧠 模块一：LLM 小说分割 & 提示词引擎

这是流水线的「大脑」——LLM 负责将长篇小说拆解为可执行的分镜脚本，输出结构化 JSON。

📋 分割策略详解

📖 小说内容类型🎬 分镜策略⏱️ 时长建议
💬 对话场景每句台词 = 1 个分镜，正反打镜头切换3-4 秒/镜
⚔️ 动作场景关键动作帧，快速切换2-3 秒/镜
🌄 环境描写1 个慢推/全景镜头4-5 秒/镜
💭 心理活动角色面部特写 + 旁白3-4 秒/镜
⚡ 回忆/闪回降低饱和度，柔焦处理2-3 秒/镜

📝 LLM 系统提示词模板

你是一个专业的 AI 短剧分镜师。将小说片段拆分为视频分镜脚本。

📋 规则：
1. 每个分镜对应一个 3-5 秒的视频片段
2. 每个分镜必须包含：场景描述、角色外貌、动作、表情、镜头运动
3. ⚠️ 角色外貌描述必须全局一致（首次出现时定义详细外貌）
4. 镜头运动：推(push)、拉(pull)、摇(pan)、移(tracking)、固定(static)、特写(close-up)
5. image_prompt 必须是英文，包含场景、人物、光影、风格关键词

📤 输出 JSON 格式：
{
  "characters": [{
    "id": "char_01",
    "name": "角色名",
    "appearance": "详细外貌描述",
    "voice_style": "声音风格"
  }],
  "scenes": [{
    "scene_id": 1,
    "shot_id": 1,
    "description": "场景中文描述",
    "characters_in_scene": ["char_01"],
    "character_action": "角色动作描述",
    "expression": "表情描述",
    "camera": "镜头运动描述",
    "mood": "情绪氛围",
    "duration_sec": 4,
    "image_prompt": "English prompt for image/video generation",
    "negative_prompt": "blurry, deformed, extra limbs",
    "motion": "camera movement description",
    "transition_to_next": "cut / fade / dissolve",
    "dialogue": "台词内容（如有）"
  }]
}

Chapter 05

🎭 模块三：角色一致性方案

这是 AI 短剧最难也最关键的部分——让同一个角色在不同镜头、不同场景下保持一致。

⚠️ 核心原则： LLM 在后续分镜中必须复用首次定义的角色外貌描述，不允许自由发挥。在 Prompt 中注入角色注册表，强制 LLM 引用。

🐍 核心代码骨架

import json
import requests
from pathlib import Path

class AIDramaPipeline:
    """小说 → AI 短剧 全自动流水线"""
    
    def __init__(self, comfyui_url="http://127.0.0.1:8188", llm_api="http://your-llm/v1"):
        self.comfyui = comfyui_url
        self.llm = llm_api
        self.character_refs = {}  # 🎭 角色参考图注册表
        self.storyboard = None
    
    # ═══════════════════════════════════════════
    # 🧠 Step 1: LLM 分割小说
    # ═══════════════════════════════════════════
    def split_novel(self, text: str, chapter: str) -> dict:
        """调用 LLM 将小说拆分为分镜 JSON"""
        
        system_prompt = """你是专业 AI 短剧分镜师。只输出 JSON。
规则：每镜 3-5 秒，角色外貌全局一致，image_prompt 英文。"""
        
        user_prompt = f"""将小说第{chapter}章拆为分镜脚本。
已定义角色（必须严格复用外貌）：
{json.dumps(self.character_refs, ensure_ascii=False, indent=2)}

小说内容：
{text}"""
        
        resp = requests.post(f"{self.llm}/chat/completions", json={
            "model": "your-model",
            "messages": [
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            "temperature": 0.7,
            "response_format": {"type": "json_object"}
        })
        
        content = resp.json()["choices"][0]["message"]["content"]
        self.storyboard = json.loads(content)
        
        # 📝 更新角色注册表
        for char in self.storyboard.get("characters", []):
            if char["id"] not in self.character_refs:
                self.character_refs[char["id"]] = char
        
        return self.storyboard
    
    # ═══════════════════════════════════════════
    # 🎨 Step 2: 生成角色设定图
    # ═══════════════════════════════════════════
    def generate_character_refs(self):
        """为每个角色生成多角度参考图"""
        for char_id, char in self.character_refs.items():
            if "ref_images" in char:
                continue  # 已有参考图，跳过
            
            prompt = f"{char['appearance']}, character design sheet, "
            prompt += "front view, side view, 3/4 view, white background, "
            prompt += "high quality, detailed, anime style"
            
            ref_path = self._generate_image(prompt, seed=42)
            char["ref_images"] = [ref_path]
            print(f"  🎭 {char['name']} 设定图已生成: {ref_path}")
    
    # ═══════════════════════════════════════════
    # 🎞️ Step 3: 逐镜头生成视频
    # ═══════════════════════════════════════════
    def generate_all_shots(self, output_dir="./shots"):
        """遍历分镜，逐个生成视频片段"""
        Path(output_dir).mkdir(parents=True, exist_ok=True)
        shots = self.storyboard["scenes"]
        paths = []
        
        for i, shot in enumerate(shots):
            print(f"  🎬 [{i+1}/{len(shots)}] {shot['description'][:40]}...")
            
            # 构建 ComfyUI 参数
            params = {
                "positive_prompt": shot["image_prompt"],
                "negative_prompt": shot.get("negative_prompt", ""),
                "seed": -1,
                "steps": 25,
            }
            
            # 🎭 注入角色参考图
            if shot.get("characters_in_scene"):
                for cid in shot["characters_in_scene"]:
                    if cid in self.character_refs:
                        ref = self.character_refs[cid]["ref_images"][0]
                        params["ipadapter_image"] = ref
            
            # 🎬 注入镜头运动
            motion = shot.get("motion", "static")
            params["motion_lora"] = self._map_motion(motion)
            
            path = self._generate_video(params, f"{output_dir}/shot_{i:04d}.mp4")
            paths.append(path)
        
        return paths
    
    # ═══════════════════════════════════════════
    # 🔧 Step 4: 后处理
    # ═══════════════════════════════════════════
    def postprocess(self, video_paths):
        """补帧 + 超分"""
        processed = []
        for path in video_paths:
            # RIFE 补帧到 24fps
            rife_path = self._rife_interpolate(path, target_fps=24)
            # 超分辨率
            hd_path = self._upscale(rife_path)
            processed.append(hd_path)
        return processed
    
    # ═══════════════════════════════════════════
    # 🚀 一键执行
    # ═══════════════════════════════════════════
    def produce(self, novel_path: str, chapter: str):
        """完整流水线一键执行"""
        
        print("📖 读取小说...")
        with open(novel_path) as f:
            text = f.read()
        
        print("🧠 LLM 分镜中...")
        self.split_novel(text, chapter)
        
        print("🎨 生成角色设定图...")
        self.generate_character_refs()
        
        print("🎞️ 生成视频片段...")
        shots = self.generate_all_shots()
        
        print("🔧 后处理中...")
        final_shots = self.postprocess(shots)
        
        print("🎬 合并成片...")
        output = self._concatenate(final_shots)
        
        print(f"✅ 完成！输出: {output}")
        return output
