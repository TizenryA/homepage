🧠 模块一：LLM 小说分割 & 提示词引擎

流水线的「大脑」——LLM 读小说，输出结构化分镜 JSON。所有 LLM 都走 OpenAI 兼容接口，换个 API 地址就能切换模型。

📋 分割策略

📖 内容类型🎬 分镜策略⏱️ 时长
💬 对话每句台词 = 1 个分镜，正反打切换3-4 秒
⚔️ 动作关键动作帧，快速切换2-3 秒
🌄 环境1 个慢推/全景镜头4-5 秒
💭 心理角色特写 + 旁白3-4 秒
⚡ 闪回降低饱和度，柔焦2-3 秒

📝 LLM System Prompt 模板

你是一个专业的 AI 短剧分镜师。将小说片段拆分为视频分镜脚本。

📋 规则：
1. 每个分镜 = 1 个 3-5 秒视频片段
2. 必须包含：场景描述、角色外貌、动作、表情、镜头运动
3. ⚠️ 角色外貌全局一致（首次定义，后续强制复用）
4. image_prompt 必须是英文，含场景+人物+光影+风格
5. image_prompt 要具体：写清楚外貌特征、服装、场景细节、光线、构图

📤 输出 JSON：

# Flux API (通过 SiliconFlow 国内代理)
import requests

def generate_image(prompt: str, negative: str = "", seed: int = -1) -> str:
    """调用 Flux API 生成图片，返回图片 URL"""
    resp = requests.post(
        "https://api.siliconflow.cn/v1/images/generations",
        headers={"Authorization": "Bearer YOUR_API_KEY"},
        json={
            "model": "black-forest-labs/FLUX.1-schnell",  # 或 FLUX.1-dev
            "prompt": prompt,
            "negative_prompt": negative,
            "image_size": "768x1024",   # 竖屏短剧
            "batch_size": 1,
            "seed": seed if seed > 0 else None,
            "num_inference_steps": 20,
        }
    )
    return resp.json()["data"][0]["url"]

🎞️ 图生视频 API 调用示例

# 可灵 Kling API
import requests, time

def generate_video(image_url: str, prompt: str, duration: int = 5) -> str:
    """图生视频：传入关键帧图片 + 运动描述 → 返回视频 URL"""
    
    # 1️⃣ 提交任务
    resp = requests.post(
        "https://api.klingai.com/v1/videos/image2video",
        headers={"Authorization": "Bearer YOUR_KLING_KEY"},
        json={
            "model_name": "kling-v1-6",
            "image": image_url,
            "prompt": prompt,           # 运动描述："camera slowly pushes in, character turns head left"
            "negative_prompt": "blurry, static, low quality",
            "duration": str(duration),   # 5 或 10 秒
            "mode": "high quality",      # 标准 / 高品质
            "camera_type": "horizontal_pan",  # 镜头运动类型
        }
    )
    task_id = resp.json()["data"]["task_id"]
    
    # 2️⃣ 轮询等待完成
    while True:
        status = requests.get(
            f"https://api.klingai.com/v1/videos/image2video/{task_id}",
            headers={"Authorization": "Bearer YOUR_KLING_KEY"}
        ).json()
        
        if status["data"]["task_status"] == "succeed":
            return status["data"]["task_result"]["videos"][0]["url"]
        elif status["data"]["task_status"] == "failed":
            raise Exception(f"生成失败: {status}")
        
        time.sleep(10)  # 等 10 秒再查

🎬 镜头控制技巧

📷

视频 Prompt 写法

视频生成的 prompt 侧重运动描述，而非静态画面：

  • camera slowly pushes in towards the character's face
  • character turns head to the left and smiles
  • camera pans right revealing a cityscape
  • rain falls, character walks forward slowly
🎛️

可灵 camera_type 参数

可灵 API 支持预设镜头运动：

  • horizontal_pan — 水平摇镜
  • vertical_pan — 垂直摇镜
  • zoom_in / zoom_out — 推拉
  • rotate — 旋转
  • fixed — 固定机位
⚠️ 注意： 图生视频效果取决于输入图质量。先用 Flux/SDXL 生成高清关键帧（768×1024 或更高），再喂给视频 API，效果远好于直接文生视频。角色参考图也用文生图 API 生成后缓存复用。

import json
import requests
import time
from pathlib import Path

class AIDramaPipeline:
    """小说 → AI 短剧 全自动流水线（在线 API 方案）"""
    
    def __init__(self):
        # ═══ API 配置 ═══
        self.llm_base = "https://api.deepseek.com/v1"        # LLM API
        self.llm_key = "sk-xxx"
        self.llm_model = "deepseek-chat"
        
        self.image_base = "https://api.siliconflow.cn/v1"     # Flux 文生图
        self.image_key = "sk-xxx"
        self.image_model = "black-forest-labs/FLUX.1-schnell"
        
        self.video_base = "https://api.klingai.com/v1"        # 可灵 图生视频
        self.video_key = "xxx"
        
        # ═══ 角色注册表 ═══
        self.character_refs = {}   # { "char_01": { "name":..., "appearance":..., "ref_image_url":... } }
        self.storyboard = None
    
    # ═══════════════════════════════════════════
    # 🧠 Step 1: LLM 分镜
    # ═══════════════════════════════════════════
    def split_novel(self, text: str) -> dict:
        system = """你是专业 AI 短剧分镜师。只输出 JSON，不要解释。
规则：
- 每镜 3-5 秒，角色外貌全局一致
- image_prompt 英文，要具体（外貌+场景+光影+构图）
- video_prompt 英文，描述运动和镜头变化
- 已定义角色必须原样复用外貌描述，不得修改"""
        
        user = f"""已定义角色：
{json.dumps(self.character_refs, ensure_ascii=False, indent=2)}

将以下小说拆为分镜脚本：
{text[:6000]}"""  # 截断防超 token
        
        resp = requests.post(f"{self.llm_base}/chat/completions",
            headers={"Authorization": f"Bearer {self.llm_key}"},
            json={
                "model": self.llm_model,
                "messages": [
                    {"role": "system", "content": system},
                    {"role": "user", "content": user}
                ],
                "temperature": 0.7,
                "response_format": {"type": "json_object"}
            })
        
        content = resp.json()["choices"][0]["message"]["content"]
        self.storyboard = json.loads(content)
        
        # 更新角色注册表
        for char in self.storyboard.get("characters", []):
            if char["id"] not in self.character_refs:
                self.character_refs[char["id"]] = char
        
        return self.storyboard
    
    # ═══════════════════════════════════════════
    # 🖼️ Step 2: 文生图（Flux API）
    # ═══════════════════════════════════════════
    def generate_image(self, prompt: str, negative: str = "") -> str:
        resp = requests.post(f"{self.image_base}/images/generations",
            headers={"Authorization": f"Bearer {self.image_key}"},
            json={
                "model": self.image_model,
                "prompt": prompt,
                "negative_prompt": negative or "blurry, deformed, bad anatomy",
                "image_size": "768x1024",
                "num_inference_steps": 20,
            })
        return resp.json()["data"][0]["url"]
    
    # ═══════════════════════════════════════════
    # 🎞️ Step 3: 图生视频（可灵 API）
    # ═══════════════════════════════════════════
    def generate_video(self, image_url: str, prompt: str, camera: str = "fixed") -> str:
        # 提交任务
        resp = requests.post(f"{self.video_base}/videos/image2video",
            headers={"Authorization": f"Bearer {self.video_key}"},
            json={
                "model_name": "kling-v1-6",
                "image": image_url,
                "prompt": prompt,
                "negative_prompt": "blurry, static, low quality",
                "duration": "5",
                "mode": "high quality",
                "camera_type": camera,
            })
        task_id = resp.json()["data"]["task_id"]
        
        # 轮询等待
        for _ in range(60):  # 最多等 10 分钟
            time.sleep(10)
            status = requests.get(
                f"{self.video_base}/videos/image2video/{task_id}",
                headers={"Authorization": f"Bearer {self.video_key}"}
            ).json()
            
            if status["data"]["task_status"] == "succeed":
                return status["data"]["task_result"]["videos"][0]["url"]
            elif status["data"]["task_status"] == "failed":
                raise Exception(f"视频生成失败: {status}")
        
        raise TimeoutError("视频生成超时")
    
    # ═══════════════════════════════════════════
    # 🎭 Step 4: 生成角色设定图
    # ═══════════════════════════════════════════
    def generate_character_refs(self):
        for cid, char in self.character_refs.items():
            if "ref_image_url" in char:
                continue
            prompt = f"character design sheet, {char['appearance']}, "
            prompt += "front view, side view, 3/4 view, white background, "
            prompt += "high quality, detailed, full body"
            url = self.generate_image(prompt)
            char["ref_image_url"] = url
            print(f"  🎭 {char['name']} 设定图: {url}")
    
    # ═══════════════════════════════════════════
    # 🚀 Step 5: 逐镜头生成
    # ═══════════════════════════════════════════
    def generate_all_shots(self, output_dir="./shots"):
        Path(output_dir).mkdir(parents=True, exist_ok=True)
        shots = self.storyboard["scenes"]
        results = []
        
        for i, shot in enumerate(shots):
            print(f"  🎬 [{i+1}/{len(shots)}] {shot['description'][:40]}...")
            
            # 2a. 文生图：生成关键帧
            img_url = self.generate_image(
                prompt=shot["image_prompt"],
                negative=shot.get("negative_prompt", "")
            )
            
            # 2b. 图生视频：关键帧 → 动态视频
            camera_map = {
                "pan": "horizontal_pan", "tilt": "vertical_pan",
                "zoom_in": "zoom_in", "zoom_out": "zoom_out",
                "push": "zoom_in", "pull": "zoom_out",
                "fixed": "fixed", "static": "fixed",
            }
            camera = camera_map.get(shot.get("camera", "fixed"), "fixed")
            
            video_url = self.generate_video(
                image_url=img_url,
                prompt=shot.get("video_prompt", shot.get("image_prompt", "")),
                camera=camera
            )
            
            # 下载视频
            path = f"{output_dir}/shot_{i:04d}.mp4"
            with open(path, "wb") as f:
                f.write(requests.get(video_url).content)
            
            results.append({"shot": shot, "image_url": img_url, "video_path": path})
            print(f"    ✅ {path}")
        
        return results
    
    # ═══════════════════════════════════════════
    # 🚀 一键执行
    # ═══════════════════════════════════════════
    def produce(self, novel_text: str):
        print("🧠 LLM 分镜中...")
        self.split_novel(novel_text)
        
        print("🎨 生成角色设定图...")
        self.generate_character_refs()
        
        print("🎞️ 逐镜头生成视频...")
        shots = self.generate_all_shots()
        
        print(f"✅ 完成！共 {len(shots)} 个镜头")
        return shots

# ═══ 使用 ═══
pipeline = AIDramaPipeline()
pipeline.produce("""
  第一章：雨夜

  雨水顺着城市的霓虹灯管滴落。林默撑着黑色雨伞，
  穿过空旷的街道。他停下脚步，抬头看着对面大楼的顶层——
  那里亮着一盏灯。
  
  "终于找到了。"他低声说，眼神变得锐利。
""")

# 把所有镜头拼接成一个视频
import subprocess, os

def concat_shots(shot_dir: str, output: str):
    shots = sorted(f for f in os.listdir(shot_dir) if f.endswith('.mp4'))
    
    # 写 ffmpeg 列表文件
    with open("filelist.txt", "w") as f:
        for shot in shots:
            f.write(f"file '{shot_dir}/{shot}'\n")
    
    # 拼接
    subprocess.run([
        "ffmpeg", "-y", "-f", "concat", "-safe", "0",
        "-i", "filelist.txt", "-c", "copy", output
    ])

concat_shots("./shots", "episode_01_raw.mp4")
